import assert from 'node:assert/strict';
import {mkdtempSync,readFileSync,existsSync,rmSync} from 'node:fs';
import {tmpdir} from 'node:os';
import {join} from 'node:path';
import {spawnSync} from 'node:child_process';
import {inventory,sourceFingerprint,repository} from '../../scripts/documentation-lib.mjs';
const copy=mkdtempSync(join(tmpdir(),'lorcana-canonical-rule03-'));
const expected='2e6c47341b421434f0efc84a15927b8ccbd07c099f087b5c3d29702678793f58';
const env=Object.fromEntries(Object.entries(process.env).filter(([key])=>!/(?:password|secret|token|credential|private_key)|^(?:PG|POSTGRES|DATABASE)/i.test(key)));
try {
  assert.ok(readFileSync(join(repository,'.local/rule-03/canonical-check.mjs')).equals(readFileSync(join(repository,'docs/validation/rule-03-canonical-check.mjs.txt'))));
  const extract=spawnSync('tar',['-xf',join(repository,'.local/rule-03/canonical.tar'),'-C',copy],{stdio:'inherit'});
  assert.equal(extract.status,0);
  assert.equal(sourceFingerprint(copy),expected);assert.equal(sourceFingerprint(repository),expected);
  const files=inventory(copy);
  for(const path of files)assert.ok(readFileSync(join(copy,path)).equals(readFileSync(join(repository,path))),'canonical/live bytes differ: '+path);
  for(const path of ['.env.local','.local','.git','node_modules'])assert.equal(existsSync(join(copy,path)),false);
  for(const args of [['install','--frozen-lockfile'],['verify:foundation']]) {
    const r=spawnSync('pnpm',args,{cwd:copy,env,stdio:'inherit'});
    assert.equal(r.error,undefined);assert.equal(r.signal,null);assert.equal(r.status,0);
  }
  console.log('PASS canonical staged Git archive: '+files.length+' files match live bytes; frozen install and complete foundation validation; source fingerprint '+expected+'. No private files/Git/dependencies/build input copied; verified tooling caches may be reused.');
} finally {rmSync(copy,{recursive:true,force:true});assert.equal(existsSync(copy),false);console.log('PASS canonical temporary source cleanup.');}
