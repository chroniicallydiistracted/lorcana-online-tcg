 if(!membership.has(hash))membership.set(hash,new Set());membership.get(hash).add(path);
}
const batch=git(['cat-file','--batch'],Buffer.from([...membership.keys()].join('\n')+'\n'));let offset=0;
for(const [hash,paths] of membership) {
 const end=batch.indexOf(10,offset),[observed,type,sizeText]=batch.subarray(offset,end).toString().split(' '),size=Number(sizeText);

 assert.equal(scanSecrets([...paths].map(path=>({path,bytes})),{knownSecrets}).length,0,'Full history known/pattern secret scan');
