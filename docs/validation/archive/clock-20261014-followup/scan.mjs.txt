import fs from 'node:fs';
import { execFileSync } from 'node:child_process';
import { privateRedactionValues, redact } from './scripts/evidence-lib.mjs';
const root=process.cwd();
const files=[...new Set([...execFileSync('git',['diff','--name-only','e36d223'],{encoding:'utf8'}).trim().split('\n'),...execFileSync('git',['ls-files','--others','--exclude-standard'],{encoding:'utf8'}).trim().split('\n')].filter(Boolean))];
let checked=0, decoded=0;
for(const path of files){
 if(!fs.existsSync(path)||!fs.statSync(path).isFile())continue;
 const raw=fs.readFileSync(path,'utf8');
 const texts=[raw];
 if(path.endsWith('.b64.txt')){texts.push(Buffer.from(raw,'base64').toString('utf8'));decoded++;}
 for(const text of texts){
  if(redact(text,{...process.env,...privateRedactionValues(root)}).redactions)throw new Error('Private-value/pattern match in '+path+'; content withheld');
  if(/gh[pousr]_[A-Za-z0-9]{30,}|github_pat_[A-Za-z0-9_]{30,}|AKIA[A-Z0-9]{16}/.test(text))throw new Error('Token-pattern match in '+path+'; content withheld');
 }
 checked++;
}
console.log(JSON.stringify({changed_public_files_checked:checked,decoded_payloads_checked:decoded,known_values_and_patterns_absent:true,limits:'Changed-file source scan only; not arbitrary-secret or full-history certification'}));
